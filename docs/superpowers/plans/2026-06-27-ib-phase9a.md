```python
"""
services/options_chain/fetcher.py

Fetch options chain for a symbol from yfinance.
Returns a DataFrame of calls + puts for the 4 nearest expiries,
with Black-Scholes Greeks computed inline.
"""
from __future__ import annotations

import logging
from datetime import date, datetime, timezone

import pandas as pd
import yfinance as yf

from libs.pricing.black_scholes import bs_greeks

logger = logging.getLogger(__name__)

_RISK_FREE_RATE = 0.05
_MAX_EXPIRIES = 4


def fetch_chain(symbol: str) -> pd.DataFrame:
    """
    Fetch options chain for symbol from yfinance.

    Returns DataFrame with columns:
        symbol, expiry (YYYYMMDD), right (C/P), strike,
        bid, ask, iv, delta, gamma, theta, vega, ts_snapshot

    Returns empty DataFrame on any error.
    """
    try:
        ticker = yf.Ticker(symbol)
        expiries = list(ticker.options)[:_MAX_EXPIRIES]
        if not expiries:
            return pd.DataFrame()

        hist = ticker.history(period="2d")
        if hist.empty:
            return pd.DataFrame()
        spot = float(hist["Close"].iloc[-1])
        today = date.today()
        now = datetime.now(timezone.utc)
        r = _RISK_FREE_RATE

        rows: list[dict] = []
        for exp_str in expiries:
            exp_date = date.fromisoformat(exp_str)
            dte = (exp_date - today).days
            tte = max(dte, 1) / 252.0
            expiry_fmt = exp_str.replace("-", "")

            chain = ticker.option_chain(exp_str)
            for right, df in [("C", chain.calls), ("P", chain.puts)]:
                for _, row in df.iterrows():
                    iv = float(row.get("impliedVolatility") or 0)
                    strike = float(row["strike"])
                    bid = float(row.get("bid") or 0)
                    ask = float(row.get("ask") or 0)

                    delta = gamma = theta = vega = None
                    if iv > 0 and tte > 0:
                        try:
                            g = bs_greeks(
                                S=spot, K=strike, T=tte,
                                r=r, sigma=iv, flag=right.lower(),
                            )
                            delta = g["delta"]
                            gamma = g["gamma"]
                            theta = g["theta"]
                            vega  = g["vega"]
                        except Exception:
                            pass

                    rows.append({
                        "symbol":       symbol,
                        "expiry":       expiry_fmt,
                        "right":        right,
                        "strike":       strike,
                        "bid":          bid,
                        "ask":          ask,
                        "iv":           iv,
                        "delta":        delta,
                        "gamma":        gamma,
                        "theta":        theta,
                        "vega":         vega,
                        "ts_snapshot":  now,
                    })

        return pd.DataFrame(rows)

    except Exception:
        logger.exception("fetch_chain failed for %s", symbol)
        return pd.DataFrame()
```

"""
services/options_chain/ingest.py

APScheduler service: fetches options chains from yfinance twice daily
(09:25 open / 16:05 close ET) and upserts into trading.options_chain.

Run as a standalone process:
    python -m services.options_chain.ingest
"""
from __future__ import annotations

import logging
import os

import psycopg2
from apscheduler.schedulers.blocking import BlockingScheduler
from apscheduler.triggers.cron import CronTrigger

from services.options_chain.fetcher import fetch_chain

logger = logging.getLogger(__name__)


def _get_active_symbols(conn) -> list[str]:
    """Return distinct symbols from enabled strategy configs."""
    with conn.cursor() as cur:
        cur.execute(
            "SELECT DISTINCT symbol FROM trading.strategy_config WHERE enabled = TRUE"
        )
        return [row[0] for row in cur.fetchall()]


def ingest_all(conn) -> int:
    """
    Fetch and store options chains for all enabled-strategy symbols.

    For each symbol:
      - DELETE today's existing snapshot rows
      - INSERT fresh rows from fetch_chain()
    Returns total rows inserted.
    """
    symbols = _get_active_symbols(conn)
    total = 0

    for symbol in symbols:
        try:
            df = fetch_chain(symbol)
            if df.empty:
                logger.warning("fetch_chain returned empty for %s — skipping", symbol)
                continue

            with conn.cursor() as cur:
                cur.execute(
                    "DELETE FROM trading.options_chain "
                    "WHERE symbol = %s AND ts_snapshot::date = CURRENT_DATE",
                    (symbol,),
                )

            with conn.cursor() as cur:
                for _, row in df.iterrows():
                    cur.execute(
                        """
                        INSERT INTO trading.options_chain
                            (symbol, expiry, "right", strike, bid, ask,
                             iv, delta, gamma, theta, vega, ts_snapshot)
                        VALUES (%(symbol)s, %(expiry)s, %(right)s, %(strike)s,
                                %(bid)s, %(ask)s, %(iv)s, %(delta)s, %(gamma)s,
                                %(theta)s, %(vega)s, %(ts_snapshot)s)
                        """,
                        {
                            "symbol":      row["symbol"],
                            "expiry":      row["expiry"],
                            "right":       row["right"],
                            "strike":      float(row["strike"]),
                            "bid":         float(row["bid"]),
                            "ask":         float(row["ask"]),
                            "iv":          float(row["iv"]) if row["iv"] is not None else None,
                            "delta":       float(row["delta"]) if row["delta"] is not None else None,
                            "gamma":       float(row["gamma"]) if row["gamma"] is not None else None,
                            "theta":       float(row["theta"]) if row["theta"] is not None else None,
                            "vega":        float(row["vega"])  if row["vega"]  is not None else None,
                            "ts_snapshot": row["ts_snapshot"],
                        },
                    )

            conn.commit()
            total += len(df)
            logger.info("options_chain: ingested %d rows for %s", len(df), symbol)

        except Exception:
            logger.exception("options_chain ingest failed for %s", symbol)

    return total


def _job_ingest() -> None:
    conn = None
    try:
        conn = psycopg2.connect(os.environ["PGURL"])
        n = ingest_all(conn)
        logger.info("options_chain ingest complete: %d total rows", n)
    except Exception:
        logger.exception("options_chain ingest job failed")
    finally:
        if conn:
            conn.close()


def build_scheduler() -> BlockingScheduler:
    tz = "America/New_York"
    sched = BlockingScheduler(timezone=tz)

    sched.add_job(
        _job_ingest,
        CronTrigger(day_of_week="mon-fri", hour=9, minute=25, timezone=tz),
        id="ingest_chains_open",
        name="Options chain ingest (pre-open)",
    )
    sched.add_job(
        _job_ingest,
        CronTrigger(day_of_week="mon-fri", hour=16, minute=5, timezone=tz),
        id="ingest_chains_close",
        name="Options chain ingest (post-close)",
    )
    return sched


if __name__ == "__main__":
    from dotenv import load_dotenv
    load_dotenv("config/.env")
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
    sched = build_scheduler()
    logger.info("Options chain ingest service started")
    sched.start()
```

def _fetch_options_chain(conn, symbol: str) -> pd.DataFrame:
    """Fetch today's options chain snapshot for symbol from trading.options_chain."""
    try:
        with conn.cursor() as cur:
            cur.execute(
                """SELECT expiry, strike, "right", bid, ask, iv, delta, gamma, theta, vega,

"""
services/broker_service/pnl_calculator.py

Compute per-strategy daily P&L from paper_fills and trading.trades.
Upserts results into trading.strategy_pnl.
"""
from __future__ import annotations

import logging
from datetime import date
from typing import Optional

from libs.risk.performance import sharpe, max_drawdown

logger = logging.getLogger(__name__)


def _compute_pnl_for_strategy(
    fills: list,
    last_close: Optional[float],
) -> tuple[float, float, float, float]:
    """
    Compute realized P&L, unrealized P&L, net_qty, and avg_cost from a list of fills.

    Each fill is a tuple: (strategy_id, symbol, side, quantity, fill_price, ts_fill)

    Returns (realized_pnl, unrealized_pnl, net_qty, avg_cost).
    """
    position = 0.0    # signed net quantity
    avg_cost = 0.0
    realized = 0.0

    for fill in fills:
        _, _, side, qty_raw, price_raw, _ = fill
        qty = float(qty_raw)
        fp = float(price_raw)
        is_buy = str(side).upper() == "BUY"
        signed = qty if is_buy else -qty

        # Closing / reducing an existing position
        if position != 0.0 and ((position > 0) != (signed > 0)):
            closing_qty = min(abs(signed), abs(position))
            if position > 0:
                realized += closing_qty * (fp - avg_cost)
            else:
                realized += closing_qty * (avg_cost - fp)

        new_position = position + signed

        # Update avg cost: only when opening or adding to a position
        if position == 0.0 or ((position > 0) == (signed > 0)):
            # Opening or adding
            total_cost = abs(position) * avg_cost + qty * fp
            denom = abs(new_position)
            avg_cost = total_cost / denom if denom != 0 else 0.0
        else:
            # Reducing: avg_cost of remaining position is unchanged
            if new_position == 0.0:
                avg_cost = 0.0

        position = new_position

    unrealized = 0.0
    if position != 0.0 and last_close is not None:
        unrealized = position * (last_close - avg_cost)

    return realized, unrealized, position, avg_cost


def _get_last_close(conn, symbol: str) -> Optional[float]:
    """Return the most recent close price for symbol from trading.bars_1d."""
    with conn.cursor() as cur:
        cur.execute(
            "SELECT close FROM trading.bars_1d WHERE symbol = %s ORDER BY ts_date DESC LIMIT 1",
            (symbol,),
        )
        row = cur.fetchone()
    return float(row[0]) if row else None


def _get_prior_pnl_rows(conn, strategy_id: str, limit: int = 30) -> list[float]:
    """Return the last `limit` total_pnl values for rolling metrics."""
    with conn.cursor() as cur:
        cur.execute(
            """SELECT total_pnl FROM trading.strategy_pnl
               WHERE strategy_id = %s
               ORDER BY date DESC LIMIT %s""",
            (strategy_id, limit),
        )
        return [float(r[0]) for r in cur.fetchall()]


def compute_daily_pnl(conn, target_date: Optional[date] = None) -> int:
    """
    Compute and upsert P&L for all enabled strategies for target_date (default: today).

    Sources:
      - trading.paper_fills (paper/shadow strategies)
      - trading.trades is not yet included (extend in future phases)

    Returns number of rows upserted.
    """
    if target_date is None:
        target_date = date.today()

    # Get all enabled strategies
    with conn.cursor() as cur:
        cur.execute(
            "SELECT DISTINCT strategy_id, symbol FROM trading.strategy_config WHERE enabled = TRUE"
        )
        strategies = cur.fetchall()

    rows_upserted = 0

    for strategy_id, symbol in strategies:
        try:
            # Fetch all paper fills for this strategy up to and including target_date
            with conn.cursor() as cur:
                cur.execute(
                    """SELECT strategy_id, symbol, side, quantity, fill_price, ts_fill
                       FROM trading.paper_fills
                       WHERE strategy_id = %s AND ts_fill::date <= %s
                       ORDER BY ts_fill ASC""",
                    (strategy_id, target_date),
                )
                fills = cur.fetchall()

            last_close = _get_last_close(conn, symbol)
            realized, unrealized, _, _ = _compute_pnl_for_strategy(fills, last_close)

            # Rolling 30-day metrics
            prior_totals = _get_prior_pnl_rows(conn, strategy_id, limit=30)
            today_total = realized + unrealized
            series = prior_totals + [today_total]

            sharpe_30d = None
            max_dd_30d = None
            if len(series) >= 2:
                try:
                    returns = [
                        (series[i] - series[i - 1]) / abs(series[i - 1])
                        if series[i - 1] != 0 else 0.0
                        for i in range(1, len(series))
                    ]
                    sharpe_30d = sharpe(returns)
                    # max_drawdown expects an equity curve; approximate with cumulative sum
                    import math
                    equity_curve = [100_000.0 + sum(series[:i+1]) for i in range(len(series))]
                    max_dd_30d = max_drawdown(equity_curve) * 100.0
                except Exception:
                    logger.exception("rolling metrics failed for %s", strategy_id)

            with conn.cursor() as cur:
                cur.execute(
                    """
                    INSERT INTO trading.strategy_pnl
                        (strategy_id, date, realized_pnl, unrealized_pnl, sharpe_30d, max_dd_30d,
                         updated_at)
                    VALUES (%s, %s, %s, %s, %s, %s, NOW())
                    ON CONFLICT (strategy_id, date) DO UPDATE SET
                        realized_pnl   = EXCLUDED.realized_pnl,
                        unrealized_pnl = EXCLUDED.unrealized_pnl,
                        sharpe_30d     = EXCLUDED.sharpe_30d,
                        max_dd_30d     = EXCLUDED.max_dd_30d,
                        updated_at     = NOW()
                    """,
                    (strategy_id, target_date, realized, unrealized, sharpe_30d, max_dd_30d),
                )

            conn.commit()
            rows_upserted += 1
            logger.info(
                "[pnl_calculator] %s %s: realized=%.2f unrealized=%.2f",
                strategy_id, target_date, realized, unrealized,
            )

        except Exception:
            logger.exception("[pnl_calculator] failed for strategy %s", strategy_id)

    return rows_upserted
```

def _get_strategy_pnl(conn, strategy_id: str, trade_date) -> float | None:
    """
    Return total P&L (realized + unrealized) from trading.strategy_pnl for the given
    strategy_id and trade_date. Returns None if no row found.
    """
    with conn.cursor() as cur:
        cur.execute(
            """SELECT realized_pnl, unrealized_pnl
               FROM trading.strategy_pnl
               WHERE strategy_id = %s AND date = %s
               LIMIT 1""",
            (strategy_id, trade_date),
        )
        row = cur.fetchone()
    if row is None:
        return None
    return float(row[0]) + float(row[1])
